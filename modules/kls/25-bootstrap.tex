\section{The bootstrap comparison theorem and the interface functional}
\label{sec:bootstrap}

\subsection{Two elementary lemmas}

\begin{lemma}[Balance pins the Cheeger ratio]
\label{lem:half}
For log-concave $\nu$, with the generalized concavity of the isoperimetric profile
\cite{Bobkov1999LogConcave,Milman2009Isoperimetric},
\[
  h_\nu=2\,I_\nu(\tfrac12),
\]
and for any $E$ with $\nu(E)=\tfrac12$ the Cheeger excess
$\bar e_0(E):=\nu^+(E)-h_\nu\min(p,q)$ equals $e_0(E)$.
\end{lemma}

\begin{proof}
The profile is symmetric ($\nu^+(E)=\nu^+(E^c)$) and concave with $I(0^+)=0$, so for
$p\le\tfrac12$, $I(p)/p$ is nonincreasing and minimized at $p=\tfrac12$; the case
$p>\tfrac12$ is symmetric. Hence $h_\nu=\inf_pI(p)/\min(p,q)=2I(\tfrac12)$, and at $p=\tfrac12$,
$h_\nu\min(p,q)=I(\tfrac12)$.
\end{proof}

\begin{definition}[Cheeger excess]
\label{def:cheeger-excess}
$\bar e_t(E)=\mu_t^+(E)-h_{\mu_t}\min(p_t,q_t)\ \ge\ e_t(E)\ \ge0$, the excess over the Cheeger
line rather than over the profile. It is the quantity that propagates through the bootstrap;
since $\bar e\ge e$, any consumption stated for $e$ is implied.
\end{definition}

\begin{lemma}[Whitening comparison]
\label{lem:whitening}
Let $\nu$ be log-concave on $\R^n$ with nondegenerate covariance $A_\nu$. Then
\[
  h_\nu\ \ge\ \frac{\hstar_n}{\lmax(A_\nu)^{1/2}} ,
\]
so that almost surely $h_{\mu_t}\ge\hstar_n\,\lmax(A_t)^{-1/2}$ for $t>0$. Moreover $\hstar_n$
is nonincreasing in $n$.
\end{lemma}

\begin{proof}
Let $\tilde\nu=A_\nu^{-1/2}{}_\#(\nu-a_\nu)$, isotropic log-concave, so
$h_{\tilde\nu}\ge\hstar_n$, and $\nu=T_\#\tilde\nu+a_\nu$ with $T=A_\nu^{1/2}$,
$\norm T_\op=\lmax^{1/2}$. For Borel $S$ and $\eps>0$,
$T^{-1}(S_\eps)\supseteq(T^{-1}S)_{\eps/\norm T_\op}$, whence
$\nu^+(S)\ge\tilde\nu^+(T^{-1}S)/\norm T_\op\ge(\hstar_n/\norm T_\op)\min(\nu(S),1-\nu(S))$.
For monotonicity: if $\nu$ is isotropic log-concave on $\R^m$, $m\le n$, then
$\nu\otimes\gamma_{n-m}$ is isotropic log-concave on $\R^n$ and cylinder sets give
$h_{\nu\otimes\gamma_{n-m}}\le h_\nu$, so $\hstar_n\le\hstar_m$.
\end{proof}

\subsection{The main inequality}

\begin{theorem}[Bootstrap comparison]
\label{thm:bootstrap}
Let $n\ge2$, $\eps\in(0,1]$, and let $\mu$ be isotropic log-concave on $\R^n$ with
$h_\mu\le(1+\eps)\hstar_n$ (a near-worst measure of its dimension). Let $E$ be \emph{any}
measurable set with $\mu(E)=\tfrac12$, $e_0=e_0(E)=\bar e_0(E)$, and let $\eta\in(0,\tfrac12)$.
Then, with $X_t=(\lmax(A_t)-1)_+$ as in \eqref{eq:interface-def}, for every $t>0$:
\begin{equation}\label{eq:bootstrap-main}
  \E\bigl[\bar e_t(E)\,\one_{\{t<\tau_\eta\}}\bigr]
  \ \le\
  e_0+h_\mu\Bigl(\frac\eps2+\eta+\frac12\,\Prob(\tau_\eta\le t)+\frac14\,\E X_t\Bigr),
\end{equation}
\begin{equation}\label{eq:bootstrap-exit}
  \Prob(\tau_\eta\le t)\ \le\ \frac1{4\eta^2}\Bigl(t+\int_0^t\E X_s\dd s\Bigr).
\end{equation}
Consequently, with $\Xi_T=\int_0^T\E X_t\dd t$,
\begin{equation}\label{eq:bootstrap-integrated}
  \int_0^T\E\bigl[\bar e_t\,\one_{\{t<\tau_\eta\}}\bigr]\dd t
  \ \le\
  T e_0+h_\mu\Bigl[\Bigl(\frac\eps2+\eta\Bigr)T
  +\frac{T^2}{16\eta^2}+\frac{T\,\Xi_T}{8\eta^2}+\frac{\Xi_T}{4}\Bigr],
\end{equation}
and, for $0<T<1/8$, the admissible choice $\eta=T^{1/3}<1/2$ together with
$\eps\le T^{1/3}$ gives the clean form
\begin{equation}\label{eq:clean-form}
  \int_0^T\E\bigl[\bar e_t\,\one_{\{t<\tau_\eta\}}\bigr]\dd t
  \ \le\ Te_0+C\,h_\mu\bigl(T^{4/3}+\Xi_T\bigr).
\end{equation}
\end{theorem}

\begin{proof}
\emph{Step 1.} By the perimeter supermartingale \eqref{eq:perimeter-supermartingale} and
Lemma~\ref{lem:half},
$\E[\mu_t^+(E)\one_{\{t<\tau_\eta\}}]\le\mu^+(E)=\tfrac{h_\mu}2+e_0$.

\emph{Step 2.} On $\{t<\tau_\eta\}$, $\min(p_t,q_t)\ge\tfrac12-\eta$. By
Lemma~\ref{lem:whitening} and the elementary bound
$\lambda^{-1/2}\ge1-\tfrac12(\lambda-1)_+$ for $\lambda>0$ (trivial when the right side is
negative),
\[
  \E\bigl[h_{\mu_t}\min(p_t,q_t)\one_{\{t<\tau_\eta\}}\bigr]
  \ \ge\
  \hstar_n\Bigl(\frac12-\eta\Bigr)\Bigl(\Prob(\tau_\eta>t)-\frac{\E X_t}2\Bigr).
\]

\emph{Step 3.} Combine with $\hstar_n\ge h_\mu/(1+\eps)\ge(1-\eps)h_\mu$ and
$\hstar_n\le h_\mu$. Writing $P=\Prob(\tau_\eta\le t)$,
\[
  \frac12-(1-\eps)\Bigl(\frac12-\eta\Bigr)(1-P)
  \ \le\ \eta+\frac P2+\frac\eps2 ,
\]
since $\tfrac12-(\tfrac12-\eta)(1-P)=\eta+(\tfrac12-\eta)P\le\eta+\tfrac P2$ and the
$\eps$-term contributes at most $\eps/2$. This yields \eqref{eq:bootstrap-main}.

\emph{Step 4.} $p$ is a martingale with $\dd[p]_t=s_tr_t\dd t$ by \eqref{eq:qv-p}, and
$s_tr_t\le\tfrac14\lmax(A_t)\le\tfrac14(1+X_t)$ by \eqref{eq:B-le-A} and $s\le\tfrac14$.
Doob's $L^2$ inequality gives \eqref{eq:bootstrap-exit}.

\emph{Step 5.} Integrate \eqref{eq:bootstrap-main} over $[0,T]$ using
$\int_0^T\Prob(\tau_\eta\le t)\dd t\le\tfrac1{4\eta^2}(\tfrac{T^2}2+T\Xi_T)$.
\end{proof}

\begin{remark}[Features]
\label{rem:bootstrap-features}
(a) The estimate holds for every balanced cut; near-minimality enters only through the size of
$e_0$. The hypothesis is on the measure --- near-worstness --- which is exactly the hypothesis a
proof by contradiction is entitled to. (b) All randomness is compressed into the scalar
functional $\Xi_T$, and crucially $\Xi_T$ enters multiplied by $h_\mu$: in the contradiction
regime $h_\mu\approx\hstar_n$ is small, so the bootstrap tolerates covariance excursions of
size up to $\sim1/\hstar_n$ --- a genuine weakening of the operator-norm control warned about
at \eqref{eq:trivial-lambda}. (c) A pathwise variant holds with constant probability: on
$\{\tau_\eta>T\}\cap\{\sup_{t\le T}\lmax(A_t)\le1+\sigma\}\cap\{\sup_{t\le T}\mu_t^+(E)\le
(1+\theta)\mu^+(E)\}$ --- the last event of probability $\ge\theta/(1+\theta)$ by Doob's
maximal inequality for the nonnegative perimeter martingale --- the same algebra gives
$\sup_{t\le T}\bar e_t\le(1+\theta)e_0+h_\mu(\theta+\tfrac\eps2+\eta+\tfrac\sigma4)$ pathwise.
Since the final consumption only requires events of probability $\ge\tfrac12$, constant
probability may suffice for a future stability argument needing pathwise near-minimality.
\end{remark}

\subsection{Evaluating the interface functional}
\label{subsec:interface}

\begin{lemma}[Crude evaluation]
\label{lem:crude}
For every isotropic log-concave $\mu$ on $\R^n$ and $1/n\le T\le1$,
$\Xi_T(\mu)\le1+\log(nT)$.
\end{lemma}

\begin{proof}
$\Tr A_t$ has drift $-\Tr A_t^2\le0$ by \eqref{eq:cov-sde}, so
$\E\lmax(A_t)\le\E\Tr A_t\le n$; combine with the pathwise cap
$X_t\le(t^{-1}-1)_+$ from \eqref{eq:BL-cap}, so $\E X_t\le\min(n,t^{-1})$, and integrate
splitting at $t=1/n$.
\end{proof}

\begin{remark}[The crude evaluation is provably insufficient --- by Klartag's theorem]
\label{rem:insufficiency}
For \eqref{eq:clean-form} to certify even an $O(T)$ excess bound with a small constant one
needs $h_\mu\Xi_T\le CT$, which under Lemma~\ref{lem:crude} requires $h_\mu\le CT/\log n$. But,
conditional on the July 2026 version-1 preprint of Letwin,
$h_\mu\ge\hstar_n\ge c(\log n)^{-1/4}$ \cite{Letwin2026QuadraticKLS}; even the earlier published
bound $c(\log n)^{-1/2}$ \cite{Klartag2023Logarithmic} exceeds $CT/\log n$ for all large $n$.
The crude certificate therefore never applies in high dimension: known lower bounds for KLS are
themselves an obstruction to the naive bootstrap. Any useful evaluation of $\Xi_T$ must beat
$\log n$.
\end{remark}

The polylogarithmic covariance technology does beat it. The required input is the small-time
operator-norm control imported in Section~\ref{sec:covariance-tech}
(Assumption~\ref{hyp:KI}), discharged there on the published $c/\log^2 n$ window and, conditional
on the version-1 input of Theorem~\ref{thm:letwin-qcts}, on the larger $c/\log n$ window
(Corollaries~\ref{cor:KI-discharged} and \ref{cor:KI-letwin}).

\begin{corollary}[Polylog evaluation]
\label{cor:loglog}
Under Assumption~\ref{hyp:KI}, for $t_1(n)\le T\le1$,
$\Xi_T(\mu)\le C_1t_1(n)+\log(T/t_1(n))\le C(1+\log\log n)$. Hence, when additionally
$T<1/8$ and the near-worst parameter satisfies $\eps\le T^{1/3}$, every such $\mu$ and every
balanced cut $E$ satisfy, by \eqref{eq:clean-form},
\begin{equation}\label{eq:loglog-supply}
  \int_0^T\E\bigl[\bar e_t\,\one_{\{t<\tau_\eta\}}\bigr]\dd t
  \ \le\ Te_0+C\,h_\mu\bigl(T^{4/3}+1+\log\log n\bigr).
\end{equation}
\end{corollary}

\begin{proof}
On $[0,t_1]$, $\E X_t\le\E\norm{A_t}_\op\le C_1$; on $[t_1,T]$, $X_t\le t^{-1}$ pathwise.
\end{proof}

\begin{proposition}[The relative-scale ceiling]
\label{prop:ceiling}
Fix $\kappa\in(0,1]$ and a universal $T_0>0$ satisfying
\begin{equation}\label{eq:ceiling-smallness}
  9(1+\kappa)T_0\le\frac12.
\end{equation}
If $\Xi_{T_0}(\mu)\le\kappa T_0$ held for every isotropic log-concave $\mu$, the KLS conjecture
would follow directly, with no geometric input.
Consequently, a propagation estimate at relative scale ---
$\int_0^T\E\bar e_t\dd t\le\kappa h_\mu T$ with $\kappa$ small --- cannot be expected from
Theorem~\ref{thm:bootstrap} alone: certifying it through \eqref{eq:clean-form} would require
$\Xi_T\le c\kappa T$ at a sufficiently small universal time, an input already sufficient for
the conclusion of the whole program.
\end{proposition}

\begin{proof}
$\Xi_{T_0}\le\kappa T_0$ gives $\int_0^{T_0}\E\lmax(A_t)\dd t\le(1+\kappa)T_0$. By
\eqref{eq:qv-p} and \eqref{eq:trivial-lambda}, $\dd[p]_t\le\tfrac14\lmax(A_t)\dd t$, so for any
balanced cut
\[
  \Prob(\tau\le T_0)
  \le36\,\E[p]_{T_0\wedge\tau}
  \le9\int_0^{T_0}\E\lmax(A_t)\dd t
  \le9(1+\kappa)T_0\le\frac12.
\]
Lemma~\ref{lem:survival-implies-kls} applies at the universal time $T_0$ with survival
probability at least $1/2$.
\end{proof}

\begin{remark}[The supply curve and the corridor]
\label{rem:supply}
Assembling this section and the previous one, the propagation estimates available at each scale
are:
\begin{center}
\begin{tabularx}{\textwidth}{@{}>{\raggedright\arraybackslash}p{0.30\textwidth}>{\raggedright\arraybackslash}p{0.32\textwidth}Y@{}}
\toprule
Scale of the excess bound & Status & Source \\
\midrule
Absolute, $\le C(1+e_0)T$ & Unconditional, all $\mu$, all balanced $E$ &
Prop.~\ref{prop:trivial-excess}. \\
\addlinespace
Absolute with small constant, $\le Te_0+\kappa T$ & Holds for near-worst $\mu$ whenever
$h_\mu(1+\log\log n)\le c\kappa T$; conditional on Ass.~\ref{hyp:KI} &
Cor.~\ref{cor:loglog}; the universal-time near-worst extension is Question~\ref{q:taming}. \\
\addlinespace
Relative, $\le\kappa\,h_\mu T$ & Requires, as bootstrap input, $\Xi_T\le c\kappa T$,
which at a sufficiently small universal time is already KLS-sufficient for all $\mu$; open
whether near-extremality suffices for this stronger, unweighted covariance target &
Prop.~\ref{prop:ceiling}. \\
\bottomrule
\end{tabularx}
\end{center}
On the demand side, Proposition~\ref{prop:two-tail} shows the Stein-trace estimate cannot be
satisfied slice-wise with absolute-scale excess. The corridor between what can be supplied and
what must be demanded is the residual content of the route, posed precisely in
Section~\ref{sec:open}.
\end{remark}

\begin{remark}[Product measures: sanity check, and why the bootstrap is silent there]
\label{rem:products-bootstrap}
For product $\mu$ (Section~\ref{sec:models}), $A_t$ is diagonal, each entry a nonnegative
supermartingale (drift $-A^2$), so $\Prob(\sup_{s\le t}A^{(i)}_s\ge\lambda)\le1/\lambda$ by the
maximal inequality. A more careful small-time analysis of the $n$ independent variance
processes --- which we do not carry out --- suggests $\Xi_T\asymp\log\log n$ for products of
two-sided exponentials, matching Corollary~\ref{cor:loglog} and consistent with the known
observation that a single eigenvalue can reach order $\log n$, saturating the cap
\eqref{eq:BL-cap} at $t\asymp1/\log n$ \cite{Chen2021,KlartagLehec2022Polylog}. For products,
however, $h_\mu\asymp1\gg\hstar_n$ in any hypothetical bad regime, so the near-worstness
hypothesis fails and the bootstrap is vacuous --- correctly so: products are not candidate
counterexamples, and their excess propagation follows directly from
$h_{\mu_t}\ge c\,\lmax(A_t)^{-1/2}$ by tensorization (Proposition~\ref{prop:products}).
\end{remark}
