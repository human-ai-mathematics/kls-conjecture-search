\section*{Solution: optimal scalar Gaussian partial non-centring}

\begin{proposition}
Let $u\sim N(0,A)$, $\theta\mid u\sim N(u,B)$, and weight the law by
$\exp[-r(\theta-y)^2/2]$, where $A,B>0$ and $r\ge0$. In the coordinate
$z_\alpha=\theta-\alpha u$, $0\le\alpha\le1$, the precision is
\[
 H_\alpha=\begin{pmatrix}
 A^{-1}+(1-\alpha)^2B^{-1}+r\alpha^2&-(1-\alpha)B^{-1}+r\alpha\\
 -(1-\alpha)B^{-1}+r\alpha&B^{-1}+r
 \end{pmatrix}.
\]
Both the Euclidean Gaussian Poincar\'e constant and the precision condition number are uniquely
minimized at
\[
 \alpha_\star=\frac1{1+rB}.
\]
\end{proposition}

\begin{proof}
Substitute $\theta=z_\alpha+\alpha u$ into
$u^2/A+(\theta-u)^2/B+r(\theta-y)^2$ to obtain $H_\alpha$; $y$ only changes a linear term.
The coordinate map is a determinant-one shear, so $\det H_\alpha$ is independent of $\alpha$.
Its trace is
\[
 \operatorname{tr}H_\alpha
 =\text{constant}+(B^{-1}+r)\alpha^2-2B^{-1}\alpha,
\]
whose unique minimizer on $[0,1]$ is $B^{-1}/(B^{-1}+r)=1/(1+rB)$. At fixed positive
determinant, decreasing the trace increases the smaller eigenvalue and decreases the eigenvalue
ratio. Since $C_P=1/\lambda_{\min}(H_\alpha)$, both objectives have the stated unique minimizer.
\end{proof}
