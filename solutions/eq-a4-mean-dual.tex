\section*{Solution: exact unrestricted posterior-mean dual}

\begin{theorem}
Let $\pi\in\mathcal P_1(\R^d)$ and define the unrestricted mean constant over finite-entropy,
finite-first-moment laws. Then, as an extended-valued identity,
\[
 C_{\mathrm{mean},\mathrm{all}}(\pi)
 =\sup_{\norm u=1}\sup_{t\ne0}\frac{2}{t^2}
 \log\E_\pi\exp\!\left(tu^\top(\theta-\E_\pi\theta)\right).
\]
An infinite moment on the right is interpreted as $+\infty$.
\end{theorem}

\begin{proof}
Write $X_u=u^\top(\theta-\E_\pi\theta)$ and denote the right-hand side by $K$. If $K<\infty$,
the entropy inequality gives, for every admissible $q$ and every $t\in\R$,
\[
 t\E_qX_u\le\KL(q\|\pi)+\log\E_\pi e^{tX_u}
 \le\KL(q\|\pi)+\tfrac12Kt^2.
\]
Optimizing separately for the two signs of $t$ yields
$(\E_qX_u)^2\le2K\KL(q\|\pi)$. Maximizing over $u$ gives the vector-mean inequality, hence
$C_{\mathrm{mean},\mathrm{all}}\le K$.

Conversely, let $C>C_{\mathrm{mean},\mathrm{all}}$. Then
$|\E_qX_u|\le\sqrt{2C\KL(q\|\pi)}$ for every admissible $q$. The Gibbs variational formula
(first for bounded truncations, then by monotone approximation) gives
\[
 \log\E_\pi e^{tX_u}
 =\sup_{q\ll\pi}\{t\E_qX_u-\KL(q\|\pi)\}
 \le\sup_{s\ge0}\{|t|\sqrt{2Cs}-s\}=\tfrac12Ct^2.
\]
Thus $K\le C$; sending $C\downarrow C_{\mathrm{mean},\mathrm{all}}$ gives the reverse bound.
If either side is infinite, the same two implications with arbitrary finite $C$ establish the
extended-valued equality.
\end{proof}
