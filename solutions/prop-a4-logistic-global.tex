\section*{Solution: global prior-scale rigidity for Gaussian variational families}
\label{sec:sol-prop-a4-logistic-global}

\paragraph{Refined statement.}
For a collection $\mathcal A$ of admissible competitors, write
\[
 C_{\mathrm{mean},\mathcal A}(\pi)
 :=\sup_{\substack{q\in\mathcal A\\0<\KL(q\|\pi)<\infty}}
 \frac{\norm{\E_q\theta-\E_\pi\theta}^2}{2\KL(q\|\pi)},
 \qquad
 C_{\mathcal A}(\pi)
 :=\sup_{\substack{q\in\mathcal A\\0<\KL(q\|\pi)<\infty}}
 \frac{\Wtwo^2(q,\pi)}{2\KL(q\|\pi)}.
\]
Here ``all'' in the mean constant denotes all finite-entropy, finite-first-moment laws, and
$\CTCI(\pi)$ is the optimal unrestricted $T_2$ constant in the normalization
$\Wtwo^2(q,\pi)\le 2\CTCI(\pi)\KL(q\|\pi)$.

\begin{theorem}[Exact form of Proposition~\ref{prop:a4-logistic-global}]
\label{thm:sol-prop-a4-logistic-global}
Let $\Sigma_0\succ0$, $\mu_0\in\R^d$, and let $\pi$ be the posterior associated with a finite
binary-logistic data set with finite covariates and the prior $N(\mu_0,\Sigma_0)$.  Let
$\mathcal Q$ be a variational family of nondegenerate Gaussian laws.  If, for one fixed
$S\succ0$,
\[
 \{N(a,S):a\in\R^d\}\subseteq\mathcal Q,
\]
then
\[
 C_{\mathrm{mean},\mathcal Q}(\pi)
 =C_{\mathcal Q}(\pi)
 =C_{\mathrm{mean},\mathrm{all}}(\pi)
 =\CTCI(\pi)
 =\lambda_{\max}(\Sigma_0).
\]
\end{theorem}

\begin{proof}
Write $A=\Sigma_0^{-1}$ and express the posterior density as
\begin{equation}\label{eq:sol-a4-logistic-posterior}
 \pi(\dd\theta)=Z^{-1}\exp\!\left[-\frac12
 (\theta-\mu_0)^\top A(\theta-\mu_0)-L(\theta)\right]\dd\theta,
\end{equation}
where, for observations $y_i\in\{0,1\}$,
\[
 L(\theta)=\sum_{i=1}^n
 \left\{\log(1+e^{x_i^\top\theta})-y_i x_i^\top\theta\right\}.
\]
Each summand is nonnegative and at most $\log 2+\abs{x_i^\top\theta}$.  Thus, for finite
constants $a,b\ge0$,
\begin{equation}\label{eq:sol-a4-logistic-linear-growth}
 0\le L(\theta)\le a+b\norm\theta.
\end{equation}
In particular, $0<Z<\infty$, the law $\pi$ has a small quadratic exponential moment (and hence
moments of every polynomial order), and the exact unrestricted mean dual of
Theorem~\ref{thm:a4-mean-dual} applies.

Put $\lambda=\lambda_{\max}(\Sigma_0)$.  The certified finite-logistic rigidity result,
Proposition~\ref{prop:a2-logistic-global}, gives
\begin{equation}\label{eq:sol-a4-logistic-t2-upper}
 \CTCI(\pi)=\lambda.
\end{equation}
The entropy inequality, tested with a sufficiently small positive multiple of
$\norm\theta^2$, shows that every finite-entropy competitor has a finite second moment.  For
every such competitor, Jensen's inequality under any coupling gives
\[
 \norm{\E_q\theta-\E_\pi\theta}^2\le \Wtwo^2(q,\pi),
\]
and \eqref{eq:sol-a4-logistic-t2-upper} then bounds the right-hand side by
$2\lambda\KL(q\|\pi)$.  For the unrestricted mean class, the same conclusion (including its
extended-value domain) is exactly the mean inequality identified with the directional
log-MGF envelope by Theorem~\ref{thm:a4-mean-dual}.  Consequently,
\begin{equation}\label{eq:sol-a4-logistic-upper-chain}
 C_{\mathrm{mean},\mathcal Q}\le C_{\mathcal Q}\le\lambda,
 \qquad
 C_{\mathrm{mean},\mathcal Q}\le C_{\mathrm{mean},\mathrm{all}}\le\lambda.
\end{equation}

It remains to prove one matching lower bound.  Fix an arbitrary $v\in\R^d\setminus\{0\}$ and,
using the location-rich hypothesis, set
\[
 q_{t,v}:=N(\mu_0+tv,S)\in\mathcal Q,
 \qquad t>0.
\]
The Gaussian entropy is independent of $t$, while the prior part of the reverse KL is exact:
\[
 \frac12\E_{q_{t,v}}
 \big[(\theta-\mu_0)^\top A(\theta-\mu_0)\big]
 =\frac12t^2v^\top Av+\frac12\operatorname{tr}(AS).
\]
If $G\sim N(0,S)$, the linear-growth estimate \eqref{eq:sol-a4-logistic-linear-growth} gives
\[
 0\le\E_{q_{t,v}}L
 \le a+b\big(\norm{\mu_0+tv}+\E\norm G\big)=O(1+t).
\]
Expanding $\KL(q_{t,v}\|\pi)$ from \eqref{eq:sol-a4-logistic-posterior}, all remaining terms
are independent of $t$.  Hence
\begin{equation}\label{eq:sol-a4-logistic-kl-asymptotic}
 2\KL(q_{t,v}\|\pi)=t^2v^\top Av+O(1+t).
\end{equation}
Writing $\bar\theta=\E_\pi\theta$, the corresponding mean displacement satisfies
\begin{equation}\label{eq:sol-a4-logistic-mean-asymptotic}
 \norm{\E_{q_{t,v}}\theta-\bar\theta}^2
 =\norm{\mu_0+tv-\bar\theta}^2
 =t^2\norm v^2+O(1+t).
\end{equation}
Therefore every nonzero direction has the exact limiting mean ratio
\begin{equation}\label{eq:sol-a4-logistic-directional-ratio}
 \lim_{t\to\infty}
 \frac{\norm{\E_{q_{t,v}}\theta-\E_\pi\theta}^2}
      {2\KL(q_{t,v}\|\pi)}
 =\frac{\norm v^2}{v^\top\Sigma_0^{-1}v}.
\end{equation}
Moreover, for every coupling of $q_{t,v}$ and $\pi$, Jensen's inequality yields
\begin{equation}\label{eq:sol-a4-logistic-w2-mean-lower}
 \Wtwo^2(q_{t,v},\pi)
 \ge\norm{\E_{q_{t,v}}\theta-\E_\pi\theta}^2.
\end{equation}
This verifies the transport lower bound without making any covariance-matching assumption on
$S$.

Finally, the Rayleigh optimization is
\begin{equation}\label{eq:sol-a4-logistic-rayleigh}
 \sup_{v\ne0}\frac{\norm v^2}{v^\top\Sigma_0^{-1}v}
 =\frac1{\lambda_{\min}(\Sigma_0^{-1})}
 =\lambda_{\max}(\Sigma_0)=\lambda,
\end{equation}
and it is attained by every top eigenvector of $\Sigma_0$.  Equations
\eqref{eq:sol-a4-logistic-directional-ratio}--\eqref{eq:sol-a4-logistic-rayleigh} imply
$C_{\mathrm{mean},\mathcal Q}\ge\lambda$; by
\eqref{eq:sol-a4-logistic-w2-mean-lower} they also imply $C_{\mathcal Q}\ge\lambda$.  Along a
top eigendirection, the pointwise $T_2(\lambda)$ upper bound and the mean lower bound squeeze the
transport ratio itself to $\lambda$.  Combining these lower bounds with
\eqref{eq:sol-a4-logistic-upper-chain} and \eqref{eq:sol-a4-logistic-t2-upper} gives the two exact
chains
\[
 \lambda\le C_{\mathrm{mean},\mathcal Q}\le C_{\mathcal Q}
 \le\CTCI=\lambda,
 \qquad
 \lambda\le C_{\mathrm{mean},\mathcal Q}\le C_{\mathrm{mean},\mathrm{all}}
 \le\CTCI=\lambda,
\]
which prove every asserted equality.
\end{proof}

\paragraph{Obstructions respected.}
The ledger node has the formal \texttt{bounded\_by} edges \texttt{obs:flat-direction} and
\texttt{obs:gaussian-tail-rigidity}.  Those and the neighboring A4-wide fences are respected:
\begin{itemize}[leftmargin=*]
 \item \path{obs:flat-direction} and \path{obs:gaussian-tail-rigidity}: the remote translations
 explicitly realize the prior scale rather than claim a global posterior/Fisher scale.
 \item \path{obs:heavy-tail-no-classical} and \path{obs:restricted-not-finite}: the target has
 quadratic Gaussian prior tails, and the proved global $T_2$ bound makes the restricted constant
 finite; no heavy-tailed or KL-cutoff-only conclusion is asserted.
 \item \path{obs:symmetry-vs-physical}: no multimodal, spectral, or quotient conclusion is made.
\end{itemize}
This is a global rigidity result, not a localized ELBO certificate, so it neither merges
approximation and optimization errors nor assumes that the unknown normalizing constant is
computable.
